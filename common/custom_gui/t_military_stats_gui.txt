## Military Stats GUI - Made by tymshey

custom_button = { # open
    name = t_open_military_stats
    potential = {
		ai = no
		NOT = {
			has_country_flag = t_military_stats_open_flag
		}
    }
    trigger = {
		ai = no
	}
    effect = {
		hidden_effect = {
			t_export_military_stats = yes
			clr_country_flag = t_military_stats_open_flag
			set_country_flag = t_military_stats_open_flag
		}
	}
    tooltip = t_open_military_stats_tt
}

custom_button = { # close
    name = t_close_military_stats
    potential = {
		ai = no
		has_country_flag = t_military_stats_open_flag

	}
    effect = {

		}
	}

}
