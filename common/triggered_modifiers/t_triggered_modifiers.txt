# rule breaks

t_above_2_player_alliances = {
	potential = {
		ai = no
	}
	
trigger = {
calc_true_if = {
    amount = 3
    all_country = {
       ai = no
        OR = {
			OR = {
			alliance_with = ROOT
			is_subject_of  = ROOT
			}
            
            AND = {
				OR = {
				 is_subject_of_type = vassal
				  is_subject_of_type = march
				}
                overlord = {
					ai = no
					OR = {
					is_subject_of = ROOT
					 alliance_with = ROOT
						}
					}
				}
			}
		}
	}
}
	discipline = -0.1
	advisor_cost = 10
	global_unrest = 5
}

t_thirteen_colonies_buff = {
	potential = {
		ai = no
	}

	trigger = {
		num_of_provinces_owned_or_owned_by_non_sovereign_subjects_with = {
			value = 50
			colonial_region = colonial_eastern_america
			is_city = yes
		}
		capital_scope = {
			NOT = {
				OR = {
					continent = north_america
					continent = south_america
				}
			}
		}
	}

	infantry_power = 0.05
}
