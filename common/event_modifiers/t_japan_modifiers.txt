### AI

# so how this works is basically the ai kinda rethinks its life over the course of two months, because these modifiers are so absurdly overpowered
# after a first monthly tick with those modifiers active, the ai gets its thread assessment reloaded (because it obviously feels much stronger)
# after a second monthly tick this gets nuked, and the ai returns to normal (and hopefully doesnt do anything stupid again)
# modifier selection is NOT random, those are some of the ones that wont break stuff
# using both at the same time is NOT advised, but shouldnt break anything

ai_iterative_monthly_tick_decisionmaking = { # makes ai reload some economic and military processes
    fire_damage_received = -0.25
    shock_damage_received = -0.25
    morale_damage = 0.25
    reinforce_speed = 0.5
    development_cost = -0.33
    development_cost_modifier = -0.2
}
