					}

					country_event = { id = flavor_pap.13 }
				}
			}
			else_if = {
				limit = {
					NOT = { exists = MNG }

						is_emperor_of_china = yes
					}
				}
				every_country = {
					limit = { is_emperor_of_china = yes }
					country_event = { id = flavor_pap.14 }
				}
			}
			else_if = {
				limit = {
					NOT = { exists = MNG }
					any_country = {
						NOT = { is_emperor_of_china = yes }
					}
					1816 = { NOT = { owned_by = PAP } }
				}
				every_country = {
					limit = { owns = 1816 }
					country_event = { id = flavor_pap.14 }
				}
			}
			else = {
				add_adm_power = 100
			}
		}
