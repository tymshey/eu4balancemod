country_event = { # Japanese-Korean Trade Partnership
	id = t_japan_events.5
	title = t_japan_events.5.title
	desc = t_japan_events.5.desc
	picture = URBANISATION_INCIDENT_eventPicture
	
	fire_only_once = yes
	
	trigger = {
		is_year = 1500
		tag = JAP
		KOR = {
			calc_true_if = {
				all_subject_country = {
					OR = {
						is_subject_of_type = colony
						is_subject_of_type = crown_colony
						is_subject_of_type = private_enterprise
						is_subject_of_type = self_governing_colony
						is_subject_of_type = european_colony
					}
				}
				amount = 2
			}
			OR = {
				has_opinion = {
					who = ROOT
					value = 100
				}
				alliance_with = ROOT
			}
		}
	}
	
	mean_time_to_happen = {
		months = 240
	}
	
	option = { # Officialize the agreement
		name = t_japan_events.5.a
		add_country_modifier = {
			name = t_japan_korea_trade_agreement_official
			duration = -1
		}
	}

	option = { # Tax it
		name = t_japan_events.5.b
		add_years_of_income = 1
		add_country_modifier = {
			name = t_japan_korea_trade_agreement_taxed
			duration = 7300
		}
	}

	option = { # Leave it to the merchants
		name = t_japan_events.5.c
		add_country_modifier = {
			name = t_japan_korea_trade_agreement_merchant
			duration = -1
		}
	}
}
