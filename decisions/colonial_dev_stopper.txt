# DECISION 3: STOP DEVELOPMENT
# Prevents (loyal) colonial nation from obtaining further development by either
# 	- Using monarch points to develop provinces
# 	- Colonizing new provices
# 	- Promote settlement growth using colonists
# Not included: gaining development by conquering provinces
#
# Logic: a colonial overlord can stop or restart their colonial subjects development at will as long as subject is loyal. If a subject becomes disloyal, any restriction is automatically deactivated (via event)
country_decisions = {
    stop_development_alaska = {
        potential = {
			any_subject_country = {
				is_colonial_nation = yes
				capital_scope = { colonial_region = colonial_alaska }
                NOT = { has_country_modifier = colonial_dev_stop }
			} 
		}
        
		allow = {
			any_subject_country = {
				capital_scope = { colonial_region = colonial_alaska }
				NOT = { liberty_desire = 50 }
                NOT = { has_country_modifier = colonial_dev_stop }
			}
		}
        
		effect = {
			every_subject_country = {
				limit = {
					capital_scope = { colonial_region = colonial_alaska }
				}
				add_country_modifier = { 
					name = colonial_dev_stop
					duration = -1  
					hidden = no
				}
			}
		}
        
		ai_will_do = {
			factor = 0
		}
    }
