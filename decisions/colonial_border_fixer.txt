# Decision 1: CEDE "UNLAWFUL" PROVINCES TO OVERLORD
# Returns provinces to overlord which are not in same colonial region as colonial nation capital
#
# Logic:
# 	Potential: in a loyal AI colonial nation has any full province (not in-construction colony) outside of the colonial region its capital is located
# 	Allow: potential + colonial subject is loyal
#	Effect: cede all full provinces (not in-construction colonies) outside of the colonial region its capital is located to its colonial overlord and remove colonial nation cores from these provinces
country_decisions = {
	return_land_alaska = {
		potential = {
			any_subject_country = {
				ai = yes
				is_colonial_nation = yes
				capital_scope = { colonial_region = colonial_alaska }
				any_owned_province = {
					NOT = { colonial_region = colonial_alaska }
					is_colony = no
				}
			}
		}
		allow = {
			any_subject_country = {
				ai = yes
				is_colonial_nation = yes
				capital_scope = { colonial_region = colonial_alaska }
				any_owned_province = {
					NOT = { colonial_region = colonial_alaska }
					is_colony = no
				}
				NOT = { liberty_desire = 50 }
			}
		}
		effect = {
			every_subject_country = {
				limit = {
					ai = yes
					capital_scope = { colonial_region = colonial_alaska }
				}
				every_owned_province = {
					limit = {
						NOT = { colonial_region = colonial_alaska }
						is_colony = no
					}
					remove_core = PREV
					cede_province = ROOT
				}
			}
		}
		ai_will_do = {
			factor = 0.25
			modifier = {
				factor = 2
				ruler_has_personality = expansionist_personality
				NOT = { ruler_has_personality = obsessive_perfectionist_personality }
			}
			modifier = {
				factor = 4
				ruler_has_personality = obsessive_perfectionist_personality
			}
		}
	}
