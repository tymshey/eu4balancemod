# DECISION 2: CHANGE CAPITAL
# Changes capital of colonial nation to defined province if they own that provice
#
# Colonial Cascadia:		873 Tsinuk (Fort Astoria in Oregon, added)
# Colonial Canada: 			993 Osheaga (Montreal)
# Colonial Eastern America:	965 Manahahttan (New York)
# Colonial Louisiana:		893 Chitimacha (New Orleans)
# Colonial California:		4637 Ohlone (San Francisco, changed from 2021 Miwok)
# Colonial Mexico: 			852 Mexico
# Colonial Caribbean: 		488 Zui (Port-Au-Prince)
# Colonial Colombia: 		825 Bogota (changed from 828 Cartagena)
# Colonial Peru:			809 Lima
# Colonial La Plata:		778 Rio de la Plata 
# Colonial Brazil:			763 Rio de Janeiro 
# Colonial Australia:		1090 Eora (Sydney, added)
#
# Logic:
# 	Potential: if an AI colonial nation owns a particular province (which is not its capital) in the same colonial region as its current capital (see list above)
# 	Then change capital of the colonial region to the specified province
country_decisions = {
	change_capital_alaska = {
		potential = {
			any_subject_country = {
				is_colonial_nation = yes
				capital_scope = { colonial_region = colonial_alaska }
				owns_core_province = 873
				NOT = { capital = 873 }
			}
		}
		allow = {
			any_subject_country = {
				capital_scope = { colonial_region = colonial_alaska }
				owns_core_province = 873
				NOT = { capital = 873 }
				NOT = { liberty_desire = 50 }
			}
		}
		effect = {
			every_subject_country = {
				limit = {
					capital_scope = { colonial_region = colonial_alaska }
				}
				set_capital = 873
			}
		}
		ai_will_do = {
			factor = 0.25
			modifier = {
				factor = 2
				ruler_has_personality = expansionist_personality
				NOT = { ruler_has_personality = obsessive_perfectionist_personality }
			}
			modifier = {
				factor = 4
				ruler_has_personality = obsessive_perfectionist_personality
			}
		}
	}
